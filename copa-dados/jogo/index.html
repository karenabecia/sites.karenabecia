/** * =================================================================== * COPA DOS DADOS — BACKEND (Google Apps Script) * =================================================================== * Este código fica colado dentro do Google Sheets do jogo * (Extensões > Apps Script) e publicado como Web App. * * COMO PUBLICAR: * 1. Abra a planilha "Copa dos Dados — Banco de Dados do Jogo" * 2. Extensões > Apps Script * 3. Apague o conteúdo padrão e cole este arquivo inteiro * 4. Clique em "Implantar" > "Nova implantação" * 5. Tipo: "App da Web" * 6. Executar como: "Eu" * 7. Quem pode acessar: "Qualquer pessoa" * 8. Copie a URL gerada (termina em /exec) — é ela que entra * no index.html, tempo1.html, admin.html etc. na variável API_URL * =================================================================== */ // Nomes das abas — não mude sem mudar aqui também const ABA_USUARIOS = 'Usuarios'; const ABA_PROGRESSO = 'Progresso'; const ABA_CONFIG = 'Configuracoes'; const ABA_FIGURINHAS = 'Figurinhas'; const ABA_RESPOSTAS = 'Respostas_Detalhadas'; function getSheet(nome) { return SpreadsheetApp.getActiveSpreadsheet().getSheetByName(nome); } /** * Ponto de entrada para todas as chamadas do front-end (GET e POST). * Toda ação é decidida pelo parâmetro "acao". */ function doPost(e) { return handleRequest(e); } function doGet(e) { return handleRequest(e); } /** * Extrai os parâmetros da requisição, cobrindo dois formatos possíveis: * 1. e.parameter — preenchido automaticamente em GET, ou em POST com * Content-Type "application/x-www-form-urlencoded". * 2. e.postData.contents — usado quando o front-end manda POST com * Content-Type "text/plain" (escolha proposital para evitar preflight * CORS e perda de corpo em redirects). Nesse caso o Apps Script NÃO * faz o parse automático, então fazemos manualmente aqui. */ function _extrairParametros(e) { if (e && e.parameter && Object.keys(e.parameter).length > 0) { return e.parameter; } if (e && e.postData && e.postData.contents) { const params = {}; const partes = e.postData.contents.split('&'); partes.forEach(parte => { const idx = parte.indexOf('='); if (idx === -1) return; const chave = decodeURIComponent(parte.substring(0, idx).replace(/\+/g, ' ')); const valor = decodeURIComponent(parte.substring(idx + 1).replace(/\+/g, ' ')); params[chave] = valor; }); return params; } return null; } function handleRequest(e) { let resultado; try { const params = _extrairParametros(e); const acao = params ? params.acao : null; if (!params || !acao) { resultado = { sucesso: false, erro: 'Nenhum parâmetro recebido. Verifique se a URL da API está correta no HTML.' }; return ContentService .createTextOutput(JSON.stringify(resultado)) .setMimeType(ContentService.MimeType.JSON); } switch (acao) { case 'cadastrar': resultado = cadastrarUsuario(params); break; case 'login': resultado = fazerLogin(params); break; case 'pegarProgresso': resultado = pegarProgresso(params); break; case 'salvarResposta': resultado = salvarResposta(params); break; case 'emitirCertificado': resultado = emitirCertificado(params); break; case 'pegarConfiguracoes': resultado = pegarConfiguracoes(); break; case 'loginAdmin': resultado = loginAdmin(params); break; case 'liberarTempo': resultado = liberarTempo(params); break; case 'listarAlunos': resultado = listarAlunos(params); break; default: resultado = { sucesso: false, erro: 'Ação não reconhecida.' }; } } catch (err) { resultado = { sucesso: false, erro: 'Erro no servidor: ' + err.message }; } return ContentService .createTextOutput(JSON.stringify(resultado)) .setMimeType(ContentService.MimeType.JSON); } function normalizarEmail(email) { return String(email || '').trim().toLowerCase(); } function normalizarTelefone(telefone) { return String(telefone || '').replace(/\D/g, ''); // só números } /** * CADASTRO — cria o usuário se o e-mail ainda não existir, * e cria a linha de progresso vazia correspondente. */ function cadastrarUsuario(params) { const nome = String(params.nome || '').trim(); const email = normalizarEmail(params.email); const telefone = normalizarTelefone(params.telefone); if (!nome || !email || !telefone) { return { sucesso: false, erro: 'Preencha nome, e-mail e telefone.' }; } const sheetUsuarios = getSheet(ABA_USUARIOS); const dados = sheetUsuarios.getDataRange().getValues(); for (let i = 1; i < dados.length; i++) { if (normalizarEmail(dados[i][1]) === email) { // Já cadastrado — apenas confirma o login em vez de duplicar return fazerLogin({ email: email, senha: telefone }); } } sheetUsuarios.appendRow([nome, email, telefone, new Date()]); const sheetProgresso = getSheet(ABA_PROGRESSO); sheetProgresso.appendRow([ email, '', '', '', '', // Tempo1 rodadas 1-4 '', '', '', '', // Tempo2 rodadas 1-4 '', '', '', '', // Tempo3 rodadas 1-4 '', // Figurinhas 'NAO', // CertificadoEmitido new Date() ]); return { sucesso: true, nome: nome, email: email }; } /** * LOGIN — e-mail + telefone (somente números) como senha. */ function fazerLogin(params) { const email = normalizarEmail(params.email); const senha = normalizarTelefone(params.senha); const sheetUsuarios = getSheet(ABA_USUARIOS); const dados = sheetUsuarios.getDataRange().getValues(); for (let i = 1; i < dados.length; i++) { const linhaEmail = normalizarEmail(dados[i][1]); const linhaTelefone = normalizarTelefone(dados[i][2]); if (linhaEmail === email && linhaTelefone === senha) { return { sucesso: true, nome: dados[i][0], email: linhaEmail }; } } return { sucesso: false, erro: 'E-mail ou senha incorretos.' }; } /** * Busca a linha de progresso de um aluno pelo e-mail. * Retorna o índice da linha (1-based) ou -1 se não achar. */ function _encontrarLinhaProgresso(sheet, email) { const dados = sheet.getDataRange().getValues(); for (let i = 1; i < dados.length; i++) { if (normalizarEmail(dados[i][0]) === email) { return i + 1; // getRange é 1-based } } return -1; } /** * Retorna o progresso completo do aluno: rodadas concluídas, * figurinhas conquistadas e status do certificado. */ function pegarProgresso(params) { const email = normalizarEmail(params.email); const sheet = getSheet(ABA_PROGRESSO); const linha = _encontrarLinhaProgresso(sheet, email); if (linha === -1) { return { sucesso: false, erro: 'Aluno não encontrado.' }; } const valores = sheet.getRange(linha, 1, 1, 16).getValues()[0]; const statusCertificado = String(valores[14] || 'NAO').toUpperCase(); return { sucesso: true, progresso: { tempo1: [valores[1], valores[2], valores[3], valores[4]], tempo2: [valores[5], valores[6], valores[7], valores[8]], tempo3: [valores[9], valores[10], valores[11], valores[12]], figurinhas: valores[13] ? String(valores[13]).split(',').filter(Boolean) : [], // 'NAO' = ainda não completou os 3 Tempos // 'SIM' = completou, pode emitir o certificado // 'EMITIDO' = já baixou o certificado, não pode de novo statusCertificado: statusCertificado, certificadoLiberado: statusCertificado === 'SIM' || statusCertificado === 'EMITIDO', certificadoJaEmitido: statusCertificado === 'EMITIDO' } }; } /** * Salva o resultado de uma rodada: marca como OK na aba Progresso, * adiciona figurinha se for a primeira vez que completa a rodada, * e registra o log detalhado na aba Respostas_Detalhadas. * * Usa LockService porque o jogo pode disparar várias chamadas em * sequência rápida (uma por rodada) — sem lock, duas execuções * concorrentes do Apps Script podem ler a mesma célula "desatualizada" * e uma sobrescrever o resultado da outra, perdendo progresso do aluno. */ function salvarResposta(params) { const lock = LockService.getScriptLock(); const conseguiuLock = lock.tryLock(10000); // espera até 10s pela vez if (!conseguiuLock) { return { sucesso: false, erro: 'O jogo está ocupado salvando outra resposta. Tente confirmar de novo em alguns segundos.' }; } try { const email = normalizarEmail(params.email); const tempo = Number(params.tempo); // 1, 2 ou 3 const rodada = Number(params.rodada); // 1 a 4 const acertou = params.acertou === 'true' || params.acertou === true; const respostaDada = String(params.respostaDada || ''); const figurinhaId = String(params.figurinhaId || ''); const sheetProgresso = getSheet(ABA_PROGRESSO); const linha = _encontrarLinhaProgresso(sheetProgresso, email); if (linha === -1) { return { sucesso: false, erro: 'Aluno não encontrado.' }; } // Log detalhado sempre é gravado, acertando ou não getSheet(ABA_RESPOSTAS).appendRow([ new Date(), email, tempo, rodada, respostaDada, acertou ? 'SIM' : 'NAO' ]); let novaFigurinha = null; if (acertou) { // Coluna da rodada: Tempo1=B-E, Tempo2=F-I, Tempo3=J-M const colunaRodada = 2 + (tempo - 1) * 4 + (rodada - 1); const celulaAtual = sheetProgresso.getRange(linha, colunaRodada).getValue(); sheetProgresso.getRange(linha, colunaRodada).setValue('OK'); SpreadsheetApp.flush(); // força a escrita antes de seguir, dentro do lock // Só dá a figurinha na primeira vez que a rodada é concluída if (celulaAtual !== 'OK' && figurinhaId) { const colunaFigurinhas = 14; const atuais = sheetProgresso.getRange(linha, colunaFigurinhas).getValue(); const lista = atuais ? String(atuais).split(',').filter(Boolean) : []; if (lista.indexOf(figurinhaId) === -1) { lista.push(figurinhaId); sheetProgresso.getRange(linha, colunaFigurinhas).setValue(lista.join(',')); novaFigurinha = figurinhaId; } } sheetProgresso.getRange(linha, 16).setValue(new Date()); // UltimaAtualizacao // Verifica se completou os 3 tempos inteiros para liberar certificado const valores = sheetProgresso.getRange(linha, 1, 1, 16).getValues()[0]; const todasRodadas = valores.slice(1, 13); const completouTudo = todasRodadas.every(v => v === 'OK'); if (completouTudo) { sheetProgresso.getRange(linha, 15).setValue('SIM'); } SpreadsheetApp.flush(); return { sucesso: true, novaFigurinha: novaFigurinha, certificadoLiberado: completouTudo }; } return { sucesso: true, novaFigurinha: null, certificadoLiberado: false }; } finally { lock.releaseLock(); } } /** * Lê a aba Configuracoes no formato em que cada CHAVE ocupa uma coluna * na linha 1, e o VALOR correspondente fica na linha 2, na mesma coluna. * Exemplo (é o formato real usado nesta planilha): * Linha 1: Tempo1_Liberado | Tempo2_Liberado | Tempo3_Liberado | SenhaAdmin * Linha 2: SIM | NAO | NAO | minhasenha * * Também aceita, como alternativa, o formato em que chave e valor ficam * na mesma linha, em colunas adjacentes (A=chave, B=valor) — para não * quebrar se a planilha for reorganizada no futuro. */ function _lerConfiguracoes() { const sheet = getSheet(ABA_CONFIG); const dados = sheet.getDataRange().getValues(); const config = {}; if (dados.length === 0) return config; const linha1 = dados[0] || []; const linha2 = dados[1] || []; // Formato principal: chave na linha 1, valor na linha 2, mesma coluna. for (let col = 0; col < linha1.length; col++) { const chave = String(linha1[col] || '').trim(); if (!chave) continue; const valor = String(linha2[col] || '').trim(); if (valor) config[chave] = valor; } // Formato alternativo: percorre as linhas a partir da 3ª (pula as // duas já usadas pelo formato principal) procurando pares chave // (coluna A) / valor (coluna B) na mesma linha — só preenche o que // ainda não foi capturado acima, para não sobrescrever por engano. for (let i = 2; i < dados.length; i++) { const chave = String(dados[i][0] || '').trim(); if (!chave || config[chave]) continue; const valorMesmaLinha = String(dados[i][1] || '').trim(); if (valorMesmaLinha) config[chave] = valorMesmaLinha; } return config; } /** * Marca o certificado como emitido (uma única vez por aluno) e salva * o nome exato que a pessoa digitou para a emissão. Usa o mesmo padrão * de lock de salvarResposta, para evitar que dois cliques rápidos no * botão de emitir gerem uma corrida e burlem a regra de "1 vez só". */ function emitirCertificado(params) { const lock = LockService.getScriptLock(); const conseguiuLock = lock.tryLock(10000); if (!conseguiuLock) { return { sucesso: false, erro: 'O jogo está ocupado. Tente novamente em alguns segundos.' }; } try { const email = normalizarEmail(params.email); const nomeParaCertificado = String(params.nomeCertificado || '').trim(); if (!nomeParaCertificado) { return { sucesso: false, erro: 'Informe o nome para o certificado.' }; } const sheet = getSheet(ABA_PROGRESSO); const linha = _encontrarLinhaProgresso(sheet, email); if (linha === -1) { return { sucesso: false, erro: 'Aluno não encontrado.' }; } const statusAtual = String(sheet.getRange(linha, 15).getValue() || '').toUpperCase(); if (statusAtual === 'EMITIDO') { return { sucesso: false, erro: 'Certificado já foi emitido anteriormente. Não é possível emitir novamente.', jaEmitido: true }; } if (statusAtual !== 'SIM') { return { sucesso: false, erro: 'Você ainda não completou os 3 Tempos para liberar o certificado.' }; } sheet.getRange(linha, 15).setValue('EMITIDO'); sheet.getRange(linha, 16).setValue(new Date()); SpreadsheetApp.flush(); return { sucesso: true, nomeUsado: nomeParaCertificado }; } finally { lock.releaseLock(); } } /** * Retorna quais tempos estão liberados — lido pelo front-end * para decidir o que mostrar/bloquear no hub do jogo. */ function pegarConfiguracoes() { const config = _lerConfiguracoes(); return { sucesso: true, tempo1Liberado: String(config['Tempo1_Liberado'] || '').toUpperCase() === 'SIM', tempo2Liberado: String(config['Tempo2_Liberado'] || '').toUpperCase() === 'SIM', tempo3Liberado: String(config['Tempo3_Liberado'] || '').toUpperCase() === 'SIM' }; } /** * Confere a senha de admin (armazenada na aba Configuracoes). */ function loginAdmin(params) { const senhaTentativa = String(params.senha || ''); const config = _lerConfiguracoes(); const senhaCorreta = String(config['SenhaAdmin'] || ''); if (senhaTentativa && senhaTentativa === senhaCorreta) { return { sucesso: true }; } return { sucesso: false, erro: 'Senha incorreta.' }; } /** * Libera ou bloqueia um Tempo específico — chamado pela tela admin. * Escreve no formato real desta planilha: a chave está na linha 1 * de uma coluna, e o valor vai na linha 2 dessa mesma coluna. */ function liberarTempo(params) { const senhaTentativa = String(params.senha || ''); const checaSenha = loginAdmin({ senha: senhaTentativa }); if (!checaSenha.sucesso) return checaSenha; const tempo = params.tempo; // "1", "2" ou "3" const novoStatus = params.status === 'SIM' ? 'SIM' : 'NAO'; const chave = 'Tempo' + tempo + '_Liberado'; const sheet = getSheet(ABA_CONFIG); const linha1 = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; for (let col = 0; col < linha1.length; col++) { if (String(linha1[col] || '').trim() === chave) { sheet.getRange(2, col + 1).setValue(novoStatus); return { sucesso: true, chave: chave, status: novoStatus }; } } return { sucesso: false, erro: 'Configuração não encontrada: ' + chave }; } /** * Lista todos os alunos com progresso resumido — usado na tela admin * para acompanhamento geral da turma. */ function listarAlunos(params) { const senhaTentativa = String(params.senha || ''); const checaSenha = loginAdmin({ senha: senhaTentativa }); if (!checaSenha.sucesso) return checaSenha; const sheetUsuarios = getSheet(ABA_USUARIOS); const sheetProgresso = getSheet(ABA_PROGRESSO); const usuarios = sheetUsuarios.getDataRange().getValues(); const progressos = sheetProgresso.getDataRange().getValues(); const mapaProgresso = {}; for (let i = 1; i < progressos.length; i++) { mapaProgresso[normalizarEmail(progressos[i][0])] = progressos[i]; } const lista = []; for (let i = 1; i < usuarios.length; i++) { const email = normalizarEmail(usuarios[i][1]); const prog = mapaProgresso[email]; let rodadasCompletas = 0; let figurinhas = 0; let certificado = 'NAO'; if (prog) { const rodadas = prog.slice(1, 13); rodadasCompletas = rodadas.filter(v => v === 'OK').length; figurinhas = prog[13] ? String(prog[13]).split(',').filter(Boolean).length : 0; certificado = prog[14] || 'NAO'; } lista.push({ nome: usuarios[i][0], email: email, rodadasCompletas: rodadasCompletas, totalRodadas: 12, figurinhas: figurinhas, certificado: certificado }); } return { sucesso: true, alunos: lista }; }
